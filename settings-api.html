Quick start
curl · start an executionCopy
curl https://api.nalvo.dev/v1/executions \
  -H "Authorization: Bearer $NALVO_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "worker": "invoice-parser-v2",
    "input": { "batch": "2024-08-03" }
  }'
node · poll for the resultCopy
import { Nalvo } from "@nalvo/sdk";

const nalvo = new Nalvo(process.env.NALVO_KEY);

const execution = await nalvo.executions.create({
  worker: "invoice-parser-v2",
  input: { batch: "2024-08-03" }
});

const done = await nalvo.executions.wait(execution.id);
console.log(done.output);
